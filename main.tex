% \bibentry re-typesets a bibliography entry, which also re-creates that entry's
% hyperref destination (cite.<key>). The duplicate is ignored by the PDF backend,
% so \cite links would point at the first copy (the publication list or Chapter 5)
% instead of the bibliography. Suppress only that anchor while retaining links
% contained in the entry, such as its DOI.
\makeatletter
\newcommand{\bibentrynl}[1]{%
    \begingroup
    \let\hyper@natanchorstart\@gobble
    \let\hyper@natanchorend\relax
    \bibentry{#1}%
    \endgroup
}
\makeatother
