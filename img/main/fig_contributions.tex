% The same 4x3 image grid is shown twice. Only three tiles are observed;
% red intensities below are illustrative attention-entropy scores, not data.
\begin{scope}[shift={(p1.center)}]
  \def\ameimage{\node[inner sep=0pt] at (-0.75,1.0)
    {\includegraphics[width=1.6cm]{img/main/three/renders/contributions_scene.png}};}
  \fill[black!8] (-1.75,0.25) rectangle (0.25,1.75);
  % Observed tiles show recognisable scene content; all others stay masked.
  \foreach \x/\y in {0/2,0/1,1/1} {
    \begin{scope}
      \clip ({-1.75+0.5*\x},{0.25+0.5*\y}) rectangle ++(0.5,0.5);
      \ameimage
    \end{scope}
  }
  \draw[step=0.5, black!40, line width=0.3pt, shift={(-1.75,0.25)}]
    (0,0) grid (2,1.5);
  \node[font=\scriptsize, align=center] at (1.05,1.0) {Seen\\so far};

  \draw[-{Latex[length=1.4mm,width=1mm]}, thick, steelblue31119180!70!black]
    (-0.75,0.16) -- (-0.75,-0.16);
  \node[font=\scriptsize, align=center, anchor=north, crimson2143940!75!black] at (1.05,-0.25) {Attention\\entropy};

  % Higher entropy is darker; observed tiles retain their image content.
  \foreach \x/\y/\score in {0/0/12,1/0/22,2/0/70,3/0/18,
      0/1/0,1/1/0,2/1/60,3/1/95,
      0/2/0,1/2/25,2/2/40,3/2/55} {
    \fill[crimson2143940!\score!white]
      ({-1.75+0.5*\x},{-1.75+0.5*\y}) rectangle ++(0.5,0.5);
  }
  \begin{scope}[shift={(0,-2)}]
    \foreach \x/\y in {0/2,0/1,1/1} {
      \begin{scope}
        \clip ({-1.75+0.5*\x},{0.25+0.5*\y}) rectangle ++(0.5,0.5);
        \ameimage
      \end{scope}
    }
  \end{scope}
  \draw[step=0.5, black!40, line width=0.3pt, shift={(-1.75,-1.75)}]
    (0,0) grid (2,1.5);
  % Select the maximum among unseen cells, preserving its grid position.
  \draw[black!85, line width=1pt] (-0.25,-1.25) rectangle (0.25,-0.75);
  \node[font=\scriptsize] (amenext) at (1.05,-1.45) {next};
  \draw[-{Latex[length=1.2mm,width=0.9mm]}, thick, steelblue31119180!70!black]
    (amenext.west) -- (0.3,-1.0);
\end{scope}
